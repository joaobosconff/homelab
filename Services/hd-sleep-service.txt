[Unit]
Description=Configura o sleep para HDs
After=multi-user.target hdd-WD.mount hdd-BARRACUDA.mount

[Service]
Type=oneshot
ExecStart=/usr/sbin/hdparm -S 120 -B 127 /dev/disk/by-uuid/ad0f6044-9c46-4358-bc7f-b260d6572b3e
ExecStart=/usr/sbin/hdparm -S 120 /dev/disk/by-uuid/e66bd5ad-ebce-4e81-8fc6-807e649e60f5

[Install]
WantedBy=multi-user.target


---

sudo nano /etc/systemd/system/hd-sleep.service
sudo systemctl enable /etc/systemd/system/hd-sleep.service
